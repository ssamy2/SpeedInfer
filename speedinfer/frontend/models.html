RAPID INTEGRATION

Call any model with the standard OpenAI SDK

import os
from openai import OpenAI

# Simply configure your SpeedInfer credentials
client = OpenAI(
    base_url="https://speedinfer.com/v1",
    api_key=os.environ.get("SPEEDINFER_API_KEY"),
)

# Call any open foundation model or proprietary closed model
response = client.chat.completions.create(
    model="deepseek-ai/DeepSeek-V3",  # Or "meta-llama/Llama-3.3-70B-Instruct"
    messages=[
        {"role": "system", "content": "You are an expert software architect."},
        {"role": "user", "content": "Explain KV-cache reuse in high-concurrency LLM inference."}
    ],
    max_tokens=256,
    temperature=0.7,
    stream=True
)

for chunk in response:
    content = chunk.choices[0].delta.content or ""
    print(content, end="", flush=True)
